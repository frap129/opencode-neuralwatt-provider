# NeuralWatt Provider Plugin Implementation Plan

> **For agentic workers:** REQUIRED: Use subagent-driven-development (if subagents available). Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an OpenCode plugin that provides zero-config NeuralWatt provider support with dynamic model discovery, API key auth, and Qwen system message merging.

**Architecture:** Three-file module (`constants.ts`, `models.ts`, `index.ts`) exporting a single `NeuralWattPlugin` that returns `auth`, `provider`, and `experimental.chat.system.transform` hooks. Models are fetched once from the NeuralWatt API on first use, cached in-memory, and transformed into `ModelV2` objects.

**Tech Stack:** TypeScript, Bun runtime, `@opencode-ai/plugin`, `@opencode-ai/sdk/v2`, vitest

```ts
import { BASE_URL, DEFAULT_CAPABILITIES, MODEL_CAPABILITIES } from './constants.ts';
import type { Model as ModelV2 } from '@opencode-ai/sdk/v2';

export function transformModel(id: string): ModelV2 {
  const known = MODEL_CAPABILITIES[id];
  const isKnown = known !== undefined;
  const hasImageInput = isKnown && known.modalities?.input.includes('image') === true;

  const reasoning = isKnown ? !id.endsWith('-fast') : false;

  const caps = isKnown
    ? {
        temperature: true,
        reasoning,
        attachment: hasImageInput,
        toolcall: true,
        input: {
          text: true as const,
          audio: false as const,
          image: hasImageInput,
          video: false as const,
          pdf: false as const,
        },
        output: {
          text: true as const,
          audio: false as const,
          image: false as const,
          video: false as const,
          pdf: false as const,
        },
        interleaved: false as const,
      }
    : {
        temperature: true,
        reasoning: false,
        attachment: false,
        toolcall: true,
        input: {
          text: true as const,
          audio: false as const,
          image: false as const,
          video: false as const,
          pdf: false as const,
        },
        output: {
          text: true as const,
          audio: false as const,
          image: false as const,
          video: false as const,
          pdf: false as const,
        },
        interleaved: false as const,
      };

  return {
    id,
    providerID: 'neuralwatt',
    api: {
      id,
      url: BASE_URL,
      npm: '@ai-sdk/openai-compatible',
    },
    name: deriveName(id),
    capabilities: caps,
    cost: { input: 0, output: 0, cache: { read: 0, write: 0 } },
    limit: isKnown ? known.limit : DEFAULT_CAPABILITIES.limit,
    status: 'active',
    options: {},
    headers: {},
    release_date: '',
  };
}
```

- [ ] **Step 3: Implement `fetchModels`**

```ts
export async function fetchModels(
  apiKey: string,
  logWarning: (msg: string) => void
): Promise<Record<string, ModelV2>> {
  const headers = new Headers();
  headers.set('Authorization', `Bearer ${apiKey}`);
  const response = await fetch(`${BASE_URL}/models`, {
    headers,
  });

  if (!response.ok) {
    throw new Error(`Failed to fetch models: ${response.status} ${response.statusText}`);
  }

  const json = await response.json();

  if (!json || !Array.isArray(json.data)) {
    throw new Error('Invalid response: missing data array');
  }

  let invalidCount = 0;
  const models: Record<string, ModelV2> = {};

  for (const entry of json.data) {
    if (typeof entry !== 'object' || entry === null || typeof entry.id !== 'string') {
      invalidCount++;
      continue;
    }
    models[entry.id] = transformModel(entry.id);
  }

  if (invalidCount > 0) {
    logWarning(`Skipped ${invalidCount} invalid model entries`);
  }

  return models;
}
```

Note: `typeof entry !== 'object' || entry === null` guards against null/undefined/malformed entries before checking `entry.id`. Per spec, only entries with a string `id` are kept — missing and non-string `id` values are skipped and counted.

```ts
import { fetchModels } from './models.ts';

let cachedModels: Record<string, ModelV2> | null = null;
let fetchPromise: Promise<Record<string, ModelV2>> | null = null;

// In the provider hook, replace the stub:
provider: {
  id: 'neuralwatt',
  models: async (_provider, ctx) => {
    if (!ctx.auth || ctx.auth.type !== 'api') {
      return cachedModels ?? {};
    }

    if (cachedModels) {
      return cachedModels;
    }

    if (!fetchPromise) {
      fetchPromise = fetchModels(ctx.auth.key, logWarning)
        .then((models) => {
          cachedModels = models;
          return models;
        })
        .catch((err) => {
          logError('Failed to fetch models', err);
          return {} as Record<string, ModelV2>;
        })
        .finally(() => {
          fetchPromise = null;
        });
    }

    return fetchPromise;
  },
```
